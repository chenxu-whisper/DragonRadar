import hashlib
import json
import re
import subprocess
from pathlib import Path

root = Path(r"E:\ByteDance\Projects\Code\DragonRadar\.worktrees\task27-research")
directory = root / "docs/acceptance/evidence/task27"
manifest = json.loads((directory / "manifest.json").read_text(encoding="utf-8"))
for name, record in manifest["artifacts"].items():
    path = directory / name
    assert path.stat().st_size == record["bytes"], name
    assert hashlib.sha256(path.read_bytes()).hexdigest() == record["sha256"], name
for name, expected in manifest["source_files_sha256"].items():
    assert hashlib.sha256((root / name).read_bytes()).hexdigest() == expected, name
assert manifest["source_digest"] == hashlib.sha256(
    json.dumps(manifest["source_files_sha256"], sort_keys=True).encode()
).hexdigest()
head = subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=root).decode().strip()
assert head == manifest["head"]
assert sum(manifest["backend_shard_passed"]) == manifest["backend_collected"]
assert manifest["backend_skipped"] == 0
document = root / "docs/acceptance/task27-research.md"
for target in re.findall(r"\[[^\]]+\]\(([^)]+)\)", document.read_text(encoding="utf-8")):
    if "://" in target:
        continue
    path = target.split("#", 1)[0]
    if path:
        assert (document.parent / path).exists(), target
print(f"PASS: {len(manifest['artifacts'])} artifact hashes, "
      f"{len(manifest['source_files_sha256'])} source hashes, "
      f"{manifest['backend_collected']} backend nodes and every AC evidence link")
